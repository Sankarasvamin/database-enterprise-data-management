## 一、用户与权限系统

```sql
-- 境界等级（最基础，其他表依赖它）
CREATE TABLE cultivation_levels (
    level_id INT PRIMARY KEY AUTO_INCREMENT,
    level_name VARCHAR(50) NOT NULL UNIQUE,
    level_order INT NOT NULL UNIQUE,
    required_exp INT NOT NULL DEFAULT 0
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 职务体系
CREATE TABLE roles (
    role_id INT PRIMARY KEY AUTO_INCREMENT,
    role_name VARCHAR(50) NOT NULL UNIQUE,
    role_rank INT NOT NULL,-- 职务等级，数字越大权限越高
    description VARCHAR(200)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 修仙者（核心用户表）
CREATE TABLE cultivators (
    cultivator_id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    level_id INT NOT NULL,
    role_id INT NOT NULL,
    current_exp INT DEFAULT 0,
    status ENUM('活跃', '闭关', '离宗', '封禁') DEFAULT '活跃',
    join_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (level_id) REFERENCES cultivation_levels(level_id),
    FOREIGN KEY (role_id) REFERENCES roles(role_id),
    INDEX idx_level (level_id),
    INDEX idx_role (role_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 权限定义
CREATE TABLE permissions (
    permission_id INT PRIMARY KEY AUTO_INCREMENT,
    permission_name VARCHAR(100) NOT NULL UNIQUE,
    module VARCHAR(50) NOT NULL,         -- 所属模块：炼丹/任务/资源等
    action ENUM('查看', '创建', '修改', '删除', '审批') NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 角色权限绑定
CREATE TABLE role_permissions (
    role_id INT NOT NULL,
    permission_id INT NOT NULL,
    PRIMARY KEY (role_id, permission_id),
    FOREIGN KEY (role_id) REFERENCES roles(role_id) ON DELETE CASCADE,
    FOREIGN KEY (permission_id) REFERENCES permissions(permission_id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 数据访问控制（行级权限）
CREATE TABLE data_access_rules (
    rule_id INT PRIMARY KEY AUTO_INCREMENT,
    role_id INT NOT NULL,
    table_name VARCHAR(100) NOT NULL,
    condition_field VARCHAR(100),        -- 限制字段，如 cultivator_id
    condition_type ENUM('仅自己', '同职务', '全部') DEFAULT '仅自己',
    FOREIGN KEY (role_id) REFERENCES roles(role_id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 排行榜快照
CREATE TABLE rankings (
    ranking_id INT PRIMARY KEY AUTO_INCREMENT,
    cultivator_id INT NOT NULL,
    total_contribution INT NOT NULL,
    rank_position INT NOT NULL,

    INDEX idx_snapshot (snapshot_time),
    INDEX idx_rank (rank_position)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 实时排行榜视图
CREATE VIEW current_ranking AS
SELECT 
    c.cultivator_id,
    c.username,
    cl.level_name,
    r.role_name,
    ca.total_contribution,
    RANK() OVER (ORDER BY ca.total_contribution DESC) AS rank_position
FROM cultivators c
JOIN cultivation_levels cl ON c.level_id = cl.level_id
JOIN roles r ON c.role_id = r.role_id
JOIN contribution_accounts ca ON c.cultivator_id = ca.cultivator_id;
```

## 十一、核心触发器

```sql
-- 新修仙者自动创建贡献账户
DELIMITER //
CREATE TRIGGER trg_new_cultivator
AFTER INSERT ON cultivators
FOR EACH ROW
BEGIN
    INSERT INTO contribution_accounts (cultivator_id) VALUES (NEW.cultivator_id);
END//

-- 任务完成自动发放贡献值
CREATE TRIGGER trg_task_complete
AFTER UPDATE ON task_assignments
FOR EACH ROW
BEGIN
    IF NEW.status = '已完成' AND OLD.status != '已完成' THEN
        SET @reward = (SELECT reward_contribution FROM sect_tasks WHERE task_id = NEW.task_id);
        SET @name = (SELECT task_name FROM sect_tasks WHERE task_id = NEW.task_id);
      
        UPDATE contribution_accounts 
        SET total_contribution = total_contribution + @reward,
            last_updated = NOW()
        WHERE cultivator_id = NEW.cultivator_id;
        INSERT INTO contribution_logs (cultivator_id, change_amount, source_type, related_id, description)
        VALUES (NEW.cultivator_id, @reward, '任务', NEW.task_id, CONCAT('完成任务：', @name));
    END IF;
END//

-- 外交完成自动发放贡献值
CREATE TRIGGER trg_diplomacy_contribution
AFTER INSERT ON diplomacy_logs
FOR EACH ROW
BEGIN
    IF NEW.contribution_reward > 0 THEN
        UPDATE contribution_accounts
        SET total_contribution = total_contribution + NEW.contribution_reward,
            last_updated = NOW()
        WHERE cultivator_id = NEW.cultivator_id;
        INSERT INTO contribution_logs (cultivator_id, change_amount, source_type, related_id, description)
        VALUES (NEW.cultivator_id, NEW.contribution_reward, '外交', NEW.diplomacy_id, 
                CONCAT('外交事件：', NEW.event_type));
    END IF;
END//
DELIMITER ;
```
